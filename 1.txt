import os
import numpy as np
from PIL import Image

def convert_single_channel_to_rgb(input_dir, output_dir):
    """
    将单通道 png 图片（像素值为 0-6）转换为指定调色板的三通道 RGB 图片。
    """
    # 定义 0-6 对应的 RGB 调色板
    # 确保数据类型为 uint8
    palette = np.array([
        [255, 255, 255], # 0: 白色
        [255, 0, 0],     # 1: 红色
        [255, 255, 0],   # 2: 黄色
        [0, 0, 255],     # 3: 蓝色
        [159, 129, 183], # 4: 紫色
        [0, 255, 0],     # 5: 绿色
        [255, 195, 128]  # 6: 橙/肉色
    ], dtype=np.uint8)

    # 创建输出文件夹（如果不存在）
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)

    # 获取输入文件夹内所有的 png 文件
    filenames = [f for f in os.listdir(input_dir) if f.lower().endswith('.png')]

    if not filenames:
        print(f"未在输入路径 '{input_dir}' 中找到 PNG 图片。")
        return

    print(f"找到 {len(filenames)} 张图片，开始转换...")

    for filename in filenames:
        input_path = os.path.join(input_dir, filename)
        output_path = os.path.join(output_dir, filename)

        try:
            # 读取单通道图像
            with Image.open(input_path) as img:
                # 转换为单通道灰度模式（'L'模式），以防原图通道格式有偏差
                img_gray = img.convert('L')
                img_np = np.array(img_gray)

            # 检查像素值是否超出 0-6 的范围
            if img_np.max() > 6:
                # 如果存在超出 0-6 的像素（例如某些图像的背景可能是 255），
                # 可以选择将其裁剪或报错。这里使用 clip 将超出范围的值限制在 0-6。
                # 如果确信数据只有 0-6，可以去掉这一行以提高微小的效率。
                img_np = np.clip(img_np, 0, 6)

            # 使用 NumPy 的索引映射，直接将 (H, W) 映射为 (H, W, 3) 形状的 RGB 数组
            rgb_np = palette[img_np]

            # 将 NumPy 数组转回 PIL Image 并保存为三通道 RGB 图像
            rgb_img = Image.fromarray(rgb_np, mode='RGB')
            rgb_img.save(output_path)

        except Exception as e:
            print(f"处理图片 {filename} 时发生错误: {e}")

    print("转换完成！")

if __name__ == "__main__":
    # 配置你的文件夹路径
    INPUT_FOLDER = "path/to/your/input_folder"    # 替换为你的输入文件夹路径
    OUTPUT_FOLDER = "path/to/your/output_folder"  # 替换为你要保存的文件夹路径

    convert_single_channel_to_rgb(INPUT_FOLDER, OUTPUT_FOLDER)
